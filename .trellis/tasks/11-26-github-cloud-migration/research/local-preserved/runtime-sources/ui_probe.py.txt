def visit(n):
    if isinstance(n,dict):
        a=n.get('attributes',{})
        if a.get('text') or a.get('clickable')=='true': print({k:a.get(k) for k in ['text','type','bounds','clickable']})
        for k,v in n.items():
            if k!='attributes':visit(v)
    elif isinstance(n,list):
        for v in n:visit(v)
