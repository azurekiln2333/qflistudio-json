param([Parameter(Mandatory=$true)][int]$GrantEvent, [string]$NextOwner='note-offline-1022', [ValidateRange(30,110)][int]$MaxSeconds=110)
$ErrorActionPreference = 'Stop'
$jsonRoot = 'D:/MyNewProjects/qfli_json'
$jsonId = [DateTime]::UtcNow.ToString('yyyyMMddTHHmmssZ')
$jsonOut = "$jsonRoot/.trellis/tasks/10-22-offline-json-ide/research/named-$jsonId"
[IO.Directory]::CreateDirectory($jsonOut) | Out-Null
$jsonWatch = [Diagnostics.Stopwatch]::StartNew()
$jsonRecord = [ordered]@{ grant = $GrantEvent; nextOwner = $NextOwner; scope = 'real JSON Home named-workspace test and scoped analysis'; startedUtc = [DateTime]::UtcNow.ToString('o'); phases = @(); error = $null; terminal = $false }
$jsonDart = 'D:/ProgramFiles/flutter_ohos/bin/cache/dart-sdk/bin/dart.exe'
$jsonSnapshot = 'D:/ProgramFiles/flutter_ohos/bin/cache/flutter_tools.snapshot'
$jsonCode = 1
$jsonReady = $false
function Save-JsonRun {
  [IO.File]::WriteAllText("$jsonOut/result.json", ($jsonRecord | ConvertTo-Json -Depth 10), [Text.UTF8Encoding]::new($false))
}
try {
  Add-Type -Path "$jsonRoot/.trellis/.runtime/json-bounded-process.cs"
  $jsonReady = $true
  $jsonPhases = @(
    @{name='format'; args=@('format','test/json_named_workspace_test.dart'); limit=10},
    @{name='test'; args=@($jsonSnapshot,'--suppress-analytics','test','--no-pub','--concurrency=1','--reporter=expanded','--timeout','45s','test/json_named_workspace_test.dart'); limit=65},
    @{name='analyze'; args=@('analyze','test/json_named_workspace_test.dart'); limit=25}
  )
  foreach ($jsonPhase in $jsonPhases) {
    $jsonRemaining = [int][Math]::Min($jsonPhase.limit * 1000, ($MaxSeconds - 15) * 1000 - $jsonWatch.ElapsedMilliseconds)
    if ($jsonRemaining -le 0) { throw 'JSON shared window deadline exhausted.' }
    if ($jsonPhase.name -eq 'test') {
      & python "$jsonRoot/.trellis/.runtime/capture_integration.py" "$jsonOut/source-before.json"
      if ($LASTEXITCODE -ne 0) { throw 'Source capture failed.' }
    }
    $jsonRow = [ordered]@{name=$jsonPhase.name; pid=$null; exit=$null; treeEmpty=$null; timedOut=$false; residualCleanup=$false; error=$null}
    $jsonRecord.phases += $jsonRow
    $jsonCallback = [Action[int]]{param($jsonPid); $jsonRow.pid=$jsonPid; Save-JsonRun; Write-Output "JSON LIVE $($jsonPhase.name) PID=$jsonPid"}
    $jsonResult = [QfliLuaAudit.BoundedProcess]::Run($jsonDart, [string[]]$jsonPhase.args, $jsonRoot, "$jsonOut/$($jsonPhase.name).log", "$jsonOut/$($jsonPhase.name).stderr.log", $jsonRemaining, $jsonCallback)
    $jsonRow.pid=$jsonResult.RootProcessId; $jsonRow.exit=$jsonResult.RootExitCode; $jsonRow.treeEmpty=$jsonResult.TreeEmpty; $jsonRow.timedOut=$jsonResult.TimedOut; $jsonRow.residualCleanup=$jsonResult.ResidualTreeTerminated; $jsonRow.error=$jsonResult.Error
    Save-JsonRun
    if (!$jsonResult.TreeEmpty -or $jsonResult.TimedOut -or $jsonResult.Error) { throw 'Owned process did not finish cleanly.' }
    if ($jsonResult.RootExitCode -ne 0) { $jsonCode=1 } elseif ($jsonPhase.name -eq 'analyze' -and @($jsonRecord.phases | Where-Object {$_.exit -ne 0}).Count -eq 0) { $jsonCode=0 }
    if ($jsonPhase.name -eq 'format' -and $jsonResult.RootExitCode -ne 0) { throw 'Format failed.' }
  }
} catch {
  $jsonRecord.error=$_.Exception.ToString()
} finally {
  & python "$jsonRoot/.trellis/.runtime/capture_integration.py" "$jsonOut/source-after.json"
  $jsonRecord.finishedUtc=[DateTime]::UtcNow.ToString('o')
  $jsonRecord.elapsedSeconds=[Math]::Round($jsonWatch.Elapsed.TotalSeconds,3)
  $jsonRecord.exit=$jsonCode
  $jsonRecord.terminal=@($jsonRecord.phases | Where-Object {$_.pid -and $_.treeEmpty -ne $true}).Count -eq 0
  Save-JsonRun
  $jsonDisposition=if($jsonRecord.terminal){"Explicitly RELEASES host slot to $NextOwner."}else{'HOLD: owned tree terminal not confirmed; next owner must wait.'}
  $jsonMessage="JSON$GrantEvent bounded terminal exit=$jsonCode; $jsonDisposition Phases=$($jsonRecord.phases | ConvertTo-Json -Compress -Depth 4); evidence=$jsonOut/result.json. Actual-profile named Home test only; no full suite/native/HAP/device claim."
  [IO.File]::WriteAllText("$jsonOut/release.txt",$jsonMessage,[Text.UTF8Encoding]::new($false))
  if ($jsonReady) {
    $jsonTrellis=(Get-Command trellis).Source.Replace("'","''")
    $jsonPublish="& '$jsonTrellis' channel send qfli-core-release-sync --scope global --as json-offline-1022 --text-file '$jsonOut/release.txt'; exit `$LASTEXITCODE"
    $jsonEncoded=[Convert]::ToBase64String([Text.Encoding]::Unicode.GetBytes($jsonPublish))
    $jsonPublication=[QfliLuaAudit.BoundedProcess]::Run((Get-Process -Id $PID).Path,[string[]]@('-NoProfile','-NonInteractive','-EncodedCommand',$jsonEncoded),$jsonRoot,"$jsonOut/publication.log","$jsonOut/publication.stderr.log",5000,$null)
    $jsonRecord.publicationExit=$jsonPublication.RootExitCode
    Save-JsonRun
  }
  Write-Output "JSON result=$jsonOut/result.json"
}
Get-Content -LiteralPath "$jsonOut/test.log" -Tail 35 -ErrorAction SilentlyContinue
Get-Content -LiteralPath "$jsonOut/analyze.log" -Tail 15 -ErrorAction SilentlyContinue
exit $jsonCode
