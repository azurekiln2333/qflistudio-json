def as_map(value):
    if value is None:
        return {}
    if isinstance(value, list):
        return {k: None for k in value}
    if isinstance(value, dict):
        return value
    return {}


def cmp(label, left, right):
    print(label, "| left", len(left), "right", len(right),
          "| onlyL", len(set(left) - set(right)), "onlyR", len(set(right) - set(left)))

for name, doc in (("before", before), ("after", after)):

    cmp("prepared.originalAppInputs vs committed {} originalAppInputs".format(name), oai, other)
    cmp("prepared.originalAppInputs vs committed {} files".format(name), oai, as_map(doc.get("files")))
