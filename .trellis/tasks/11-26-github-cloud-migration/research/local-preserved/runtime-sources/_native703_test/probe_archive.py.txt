import hashlib
import json
import subprocess
from pathlib import Path

ROOT = Path(r"D:\MyNewProjects\qfli_json")

def sha(data):
    return hashlib.sha256(data).hexdigest()


def git_blob(path_in_repo):
    out = subprocess.run(
        ["git", "cat-file", "-p", "{}:{}".format(COMMIT, path_in_repo)],
        cwd=str(ROOT), capture_output=True,
    )
    return out.stdout if out.returncode == 0 else None


v = json.loads((ARCH / "validation.json").read_text(encoding="utf-8"))

print("=== A. declared file hashes ===")
bad = []
for rel, meta in v["files"].items():
    f = ARCH / rel
    if not f.is_file():
        bad.append((rel, "MISSING"))
        continue
    raw = f.read_bytes()
    if len(raw) != meta["bytes"] or sha(raw) != meta["sha256"]:
        bad.append((rel, "decl {}/{} actual {}/{}".format(
            meta["bytes"], meta["sha256"][:12], len(raw), sha(raw)[:12])))
print("declared", len(v["files"]), "mismatch", len(bad))
for item in bad:
    print("  MISMATCH", item[0], item[1])
print("preparedManifestSha256 == files[prepared-manifest.json]:",
      v["preparedManifestSha256"] == v["files"]["prepared-manifest.json"]["sha256"])

print("=== B. undeclared archive files ===")
declared = set(v["files"]) | {"validation.json"}
actual = {str(f.relative_to(ARCH)).replace("\\", "/") for f in ARCH.rglob("*") if f.is_file()}
for name in sorted(actual - declared):
    print("  UNDECLARED", name, (ARCH / name).stat().st_size)

print("=== C. committed 703 provenance hashes ===")
prov = v["initializationProvenance"]["evidence"]
for key in ("before", "after"):
    item = prov[key]
    blob = git_blob(item["path"])
    local = (ROOT / item["path"]).read_bytes()
    print(key, "declared", item["sha256"][:16])
    print("   worktree", sha(local)[:16], "match", sha(local) == item["sha256"], "bytes", len(local))
    print("   git blob ", (sha(blob)[:16] if blob is not None else "ABSENT"),
          "match", blob is not None and sha(blob) == item["sha256"])

print("=== D. file-map comparisons ===")


def file_map(doc):
    return {k: (m.get("bytes"), m.get("sha256")) for k, m in (doc.get("files") or {}).items()}


def cache_map(doc):
    refs = doc.get("cacheReferences")
    if isinstance(refs, dict):
        return set(refs)
    if isinstance(refs, list):
        return set(refs)
    return set()


retry_before = json.loads((ARCH / "before.json").read_text(encoding="utf-8-sig"))
retry_after = json.loads((ARCH / "after.json").read_text(encoding="utf-8-sig"))
retry_guarded = json.loads((ARCH / "guarded.json").read_text(encoding="utf-8-sig"))
retry_prepared = json.loads((ARCH / "prepared-manifest.json").read_text(encoding="utf-8-sig"))

pairs = [

    ("retry after  vs committed 703 after ", file_map(retry_after), file_map(old_after)),
    ("retry guarded vs committed 703 after", file_map(retry_guarded), file_map(old_after)),
    ("retry prepared vs committed 703 after", file_map(retry_prepared), file_map(old_after)),

for label, left, right in pairs:

    for k in sorted(only_left)[:6]:

    for k in sorted(only_right)[:6]:
        print("    only-right", k)
