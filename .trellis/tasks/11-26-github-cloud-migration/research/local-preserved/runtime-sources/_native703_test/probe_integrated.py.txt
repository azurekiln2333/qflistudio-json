def call(label, manifest, extra=None, spelled=None):
    def wrapper(path):
        dep = original(path)
        if extra is not None:
            dep["inputs"] = list(dep["inputs"]) + [extra]
            dep["inputPaths"] = list(dep["inputPaths"]) + [spelled if spelled is not None else extra]
        return dep

    live.parse_depfile = wrapper
    try:
        try:
            result = live.source_dependencies(app, core, kernel, prepared=manifest)
            print("{} -> PASS native={} ohos={}".format(
                label, result["isolatedNativePlugins"]["inputCount"], result["ohosPlugins"]["ohosCount"]))
        except live.BlockedInspection as error:
            print("{} -> BLOCKED: {}".format(label, error))
    finally:
        live.parse_depfile = original
